let int inicio; 

let int fin; 

let int total; 

 

function int sumarRango(int actual, int meta) { 

    if (actual == meta)  

        return actual; 

     

    /* Llamada recursiva sumando 1 hasta llegar a meta */ 

    return actual + sumarRango(actual + 1, meta); 

} 

 

inicio = 1; 

fin = 3; 

 

/* Calculamos 1+2+3 */ 

total = sumarRango(inicio, fin); 

 

write "El resultado de la suma recursiva es:"; 

write total;
