let int contador; 

let int maximo; 

let int estado;  

let boolean terminado; 

 

contador = 0; 

maximo = 4; 

estado = 1; /* si es 1 continua, 0 para */ 

 

while (estado == 1) { 

    let int incremento; /* Variable local dentro del bucle */ 

     

    incremento = 1; 

    contador = contador + incremento; 

     

    write "Contando..."; 

    write contador; 

 

    /* Si llegamos al maximo, cambiamos el estado para romper el bucle */ 

    if (contador == maximo) 

        estado = 0; 

} 

 

/* Prueba de operador logico OR y comparacion */ 

/* (1==1) es true */ 

terminado = (estado == 0) || (1 == 1); 

 

if (terminado == (1==1))  

    write "Proceso terminado con exito";
