let int contador; 

let int tope; 

let boolean ejecutando; 

 

contador = 0; 

tope = 5; 

 

/* ejcutando será true */ 

ejecutando = 1 == 1; 

 

write "Iniciando bucle..."; 

 

while (ejecutando) { 

    contador = contador + 1; 

    write contador; 

 

    /* condicion de parada, cuando contador iguala a tope */ 

    if (contador == tope) 

        /* ejecutando sera falso */ 

        ejecutando = 1 == 0; 

} 

 

write "Bucle terminado con exito";
